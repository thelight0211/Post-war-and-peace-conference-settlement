# Scripted AI desire for peace actions.
# Perfect Peace - Solo annexation system
# Fuhrerreich + faction losers + exclave protection + respect claims

peace_ai_desires = {
	take_our_cores = {
		peace_action_type = { take_states }
		enable = {
			ROOT = { tag = ROOT.FROM }
			ROOT.FROM.FROM.FROM = { is_core_of = ROOT.FROM }
		}
		ai_desire = 100
	}

	# === DON'T TAKE FROM NEUTRALS ===
	not_at_war_with_owner = {
		peace_action_type = { take_states liberate puppet force_government }
		enable = {
			NOT = { has_war_with = ROOT.FROM.FROM }
		}
		ai_desire = -1000
	}

	# === RESPECT INFLUENCE ZONES (on ALL countries) ===
	generic_influence_cutoff = {
		peace_action_type = { take_states puppet force_government liberate }
		enable = {
			check_variable = { global.pc_winners_array^num > 1 }
			ROOT.FROM.FROM.FROM = {
				pc_is_state_outside_influence_for_winner = ROOT
				NOT = { is_core_of = ROOT.FROM }
				NOT = { is_claimed_by = ROOT.FROM }
				NOT = { any_neighbor_state = { pc_is_state_claimed_by = ROOT.FROM } }
			}
			any_of_scopes = {
				array = global.pc_winners_array
				pc_is_winner = yes
				pc_current_score > 0
				NOT = { ROOT.FROM.FROM.FROM = { pc_is_state_outside_influence_for_winner = PREV } }
				if = {
					limit = { has_global_flag = pc_has_major_winner }
					is_major = yes
				}
				else_if = {
					limit = { has_global_flag = pc_has_non_subject_winner }
					is_subject = no
				}
			}
		}
		ai_desire = -1000
	}

	doesnt_control_no_claim = {
		peace_action_type = { take_states }
		enable = {
			ROOT = { tag = ROOT.FROM }
			ROOT.FROM.FROM.FROM = {
				NOT = { is_core_of = ROOT.FROM }
				NOT = { is_claimed_by = ROOT.FROM }
				NOT = { is_controlled_by = ROOT.FROM }
			}
		}
		ai_desire = -20
	}

	dont_grab_exclaves = {
		peace_action_type = take_states
		enable = {
			ROOT.FROM.FROM.FROM = {
				NOT = {
					OR = {
						is_claimed_by = ROOT.FROM
						any_neighbor_state = {
							OR = {
								is_owned_by = ROOT.FROM
								pc_is_state_claimed_by = ROOT.FROM
							}
						}
					}
				}
			}
		}
		ai_desire = -40
	}

	do_grab_stuff_near_us = {
		peace_action_type = take_states
		enable = {
			ROOT = { tag = ROOT.FROM }
			ROOT.FROM.FROM.FROM = {
				OR = {
					is_claimed_by = ROOT.FROM
					any_neighbor_state = { is_owned_by = ROOT.FROM }
				}
			}
		}
		ai_desire = 10
	}

	generic_occupies_state = {
		peace_action_type = { take_states puppet liberate force_government }
		enable = {
			ROOT.FROM.FROM.FROM = { NOT = { is_controlled_by = ROOT } }
		}
		ai_desire = -10
	}

	generic_occupies_player_state = {
		peace_action_type = { take_states puppet liberate force_government }
		enable = {
			ROOT.FROM.FROM.FROM = {
				NOT = { is_controlled_by = ROOT }
				controller = {
					pc_is_loser = no
					is_ai = no
				}
			}
		}
		ai_desire = -20
	}

	it_is_a_non_coastal_adjacency = {
		peace_action_type = take_states
		enable = {
			ROOT.FROM = { tag = ROOT }
			ROOT.FROM.FROM.FROM = {
				is_coastal = no
				any_neighbor_state = {
					is_owned_by = ROOT
				}
			}
		}
		ai_desire = 10
	}

	leave_player_controlled_land_until_later_in_the_process = {
		peace_action_type = { take_states puppet force_government liberate }
		enable = {
			ROOT.FROM.FROM.FROM = { controller = { is_ai = no } }
		}
		ai_desire = -10
	}

	# === BLOCK ALL NON-STATE ACTIONS ===
	do_not_puppet_liberate_or_force_government = {
		peace_action_type = { puppet liberate force_government }
		enable = {
			always = yes
		}
		ai_desire = -9999
	}

	# === FINISH FACTION LOSERS (Hungary, Denmark) - controlled only ===
	# Uses is_in_faction_with to identify non-capitulated faction members
	finish_faction_losers_controlled = {
		peace_action_type = take_states
		enable = {
			ROOT = { tag = ROOT.FROM }
			ROOT.FROM.FROM = { has_capitulated = no }
			ROOT.FROM.FROM.FROM = { is_controlled_by = ROOT }
			any_country = {
				has_capitulated = yes
				ROOT.FROM.FROM = { is_in_faction_with = PREV }
			}
		}
		ai_desire = 2000
	}

	# === FINISH FACTION LOSERS (Hungary, Denmark) - adjacent only ===
	finish_faction_losers_adjacent = {
		peace_action_type = take_states
		enable = {
			ROOT = { tag = ROOT.FROM }
			ROOT.FROM.FROM = { has_capitulated = no }
			ROOT.FROM.FROM.FROM = {
				any_neighbor_state = { is_owned_by = ROOT.FROM }
			}
			any_country = {
				has_capitulated = yes
				ROOT.FROM.FROM = { is_in_faction_with = PREV }
			}
		}
		ai_desire = 1200
	}

	# === RESPECT OTHER WINNERS' CORES AND CLAIMS ===
	respect_other_winner_claims = {
		peace_action_type = take_states
		enable = {
			ROOT = { tag = ROOT.FROM }
			ROOT.FROM.FROM.FROM = {
				NOT = { is_core_of = ROOT.FROM }
				NOT = { is_claimed_by = ROOT.FROM }
			}
			any_country = {
				pc_is_winner = yes
				NOT = { tag = ROOT }
				OR = {
					ROOT.FROM.FROM.FROM = { is_core_of = PREV }
					ROOT.FROM.FROM.FROM = { is_claimed_by = PREV }
				}
			}
		}
		ai_desire = -1000
	}
}
