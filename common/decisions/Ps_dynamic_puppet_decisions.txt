							NOT = {
								any_core_state = {
									is_capital = yes
									owner = { original_tag = FROM }
								}
							}
						}
					}
				}
				set_state_flag = valid_for_dynamic_puppet
				ROOT = { add_to_variable = { ROOT_from_states_count = 1 } }
			}

			set_variable = { combined_from_states_count = ROOT_from_states_count }
			set_variable = { ROOT_is_dominant = 1 }

			every_country = {
				limit = {
					is_in_faction_with = ROOT
					NOT = {
							capital_scope = {
								NOT = { is_core_of = PREV }
							}
					}
					NOT = { tag = ROOT }
				}
				set_temp_variable = { ally_from_states_count = 0 }
				every_owned_state = {
					limit = {
						any_country_with_core = { original_tag = FROM }
						NOT = { is_core_of = PREV }
						NOT = { is_claimed_by = PREV }
						is_fully_controlled_by = PREV
						NOT = {
							any_country_with_core = {
								is_ally_with = PREV
								NOT = { original_tag = FROM }
								NOT = {
									any_core_state = {
										is_capital = yes
										owner = { original_tag = FROM }
									}
								}
							}
						}
					}
					if = {
						limit = {
							PREV = { is_ai = yes }
						}
						set_state_flag = valid_for_dynamic_puppet
					}
					PREV = { add_to_temp_variable = { ally_from_states_count = 1 } }
				}

				if = {
					limit = {
						check_variable = { ally_from_states_count > ROOT.ROOT_from_states_count }
					}
					ROOT = { set_variable = { ROOT_is_dominant = 0 } }
				}
				else_if = {
					limit = {
						check_variable = { ally_from_states_count = ROOT.ROOT_from_states_count }
						check_variable = { ally_from_states_count > 0 }
					}
					
					set_temp_variable = { ally_power = 0 }
					set_temp_variable = { ally_armies = num_armies }
					add_to_temp_variable = { ally_power = num_of_factories }
					add_to_temp_variable = { ally_power = ally_armies }
					
					set_temp_variable = { root_power = 0 }
					ROOT = {
						set_temp_variable = { root_armies = num_armies }
						add_to_temp_variable = { root_power = num_of_factories }
						add_to_temp_variable = { root_power = root_armies }
					}
					
					if = {
						limit = {
							is_faction_leader = yes
							ROOT = { is_faction_leader = no }
						}
						ROOT = { set_variable = { ROOT_is_dominant = 0 } }
					}
					else_if = {
						limit = {
							is_faction_leader = no
							ROOT = { is_faction_leader = no }
							check_variable = { ally_power > root_power }
						}
						ROOT = { set_variable = { ROOT_is_dominant = 0 } }
					}
					else_if = {
						limit = {
							is_faction_leader = no
							ROOT = { is_faction_leader = no }
							check_variable = { ally_power = root_power }
						}
						set_temp_variable = { ally_manpower = 0 }
						add_to_temp_variable = { ally_manpower = manpower }
						
						set_temp_variable = { root_manpower = 0 }
						ROOT = {
							add_to_temp_variable = { root_manpower = manpower }
						}
						if = {
							limit = {
								check_variable = { ally_manpower > root_manpower }
							}
							ROOT = { set_variable = { ROOT_is_dominant = 0 } }
						}
					}
				}
				ROOT = { add_to_variable = { combined_from_states_count = ally_from_states_count } }
			}

			set_variable = { FROM_total_cores = 0 }
			every_state = {
				limit = { is_core_of = FROM }
				ROOT = { add_to_variable = { FROM_total_cores = 1 } }
			}
			set_variable = { puppet_ratio = combined_from_states_count }
			divide_variable = { puppet_ratio = FROM_total_cores }
			clear_variable = FROM_total_cores
			clear_variable = combined_from_states_count

			if = {
				limit = {
					check_variable = { ROOT_is_dominant = 1 }
				}
				set_country_flag = ROOT_creates_puppet
			}

			clear_variable = ROOT_from_states_count

			if = {
				limit = { has_country_flag = ROOT_creates_puppet }

				every_state = {
					limit = {
						has_state_flag = valid_for_dynamic_puppet
						any_country_with_core = {
							exists = yes
							NOT = { tag = FROM }
							NOT = { tag = ROOT }
							capital_scope = { is_core_of = FROM }
							NOT = {
								FROM = {
									capital_scope = { is_core_of = PREV.PREV }
								}
							}
						}
					}
					set_state_flag = dec_later_dynamic
				}

				create_dynamic_country = {
					original_tag = FROM
					copy_tag = FROM
					set_cosmetic_tag = FROM
					every_state = {
						limit = {
							has_state_flag = valid_for_dynamic_puppet
							NOT = { has_state_flag = dec_later_dynamic }
						}
						FROM = {
							transfer_state = PREV
							set_country_flag = is_dynamic_tag
						}
					}
					ROOT = { puppet = PREV }
					drop_cosmetic_tag = yes
					
					every_state = {
						limit = { has_state_flag = dec_later_dynamic }
						clr_state_flag = dec_later_dynamic
					}
					every_state = {
						limit = {
							has_state_flag = valid_for_dynamic_puppet
							is_core_of = FROM
							is_owned_by = FROM
							any_country_with_core = {
								NOT = {
									any_core_state = {
										is_capital = yes
										owner = { original_tag = FROM }
									}
								}
								is_ally_with = ROOT
								NOT = { tag = FROM }
							}
						}
						random_other_country = {
							limit = {
								is_ally_with = ROOT
								NOT = { tag = FROM }
								PREV = { is_core_of = PREV }
							}
							transfer_state = PREV
						}
					}

					if = {
						limit = {
							OR = {
								AND = {
									check_variable = { ROOT.puppet_ratio < 0.21 }  
									FROM = { num_of_controlled_states < 7 }        
								}                                              
								FROM = { num_of_controlled_states < 2 } 
								AND = {
									FROM = { num_of_controlled_states < 3 }
									ROOT = { check_variable = { puppet_ratio > 0.21 } }
									ROOT = { check_variable = { puppet_ratio < 0.30 } }
								}       
							} 
							ROOT = {
								check_variable = { ROOT_is_dominant = 1 }
							}                                                
							FROM = {
								OR = {
									any_neighbor_country = { original_tag = FROM }
									all_owned_state = { is_coastal = yes }
								}
							}
							any_other_country = {
								original_tag = FROM
								exists = yes
							}
						}
						if = {
							limit = { ROOT = { is_ai = yes } }
							random_other_country = {
								limit = { 
									original_tag = FROM
								}
								annex_country = { target = PREV }
							}
						}
						else = {
							set_country_flag = is_small_dynamic_puppet
							save_global_event_target_as = small_dynamic_puppet_global
							
							random_other_country = {
								limit = {
									original_tag = FROM
									exists = yes
								}
								set_country_flag = will_annex_small_puppet
								save_global_event_target_as = will_annex_puppet_global
							}
							ROOT = { country_event = { id = id.settlement.16 } }
						}
					}
				}

				clear_variable = puppet_ratio
				clear_variable = ROOT_is_dominant
				clr_country_flag = ROOT_creates_puppet
			}
			
			every_state = {
				limit = { has_state_flag = valid_for_dynamic_puppet }
				clr_state_flag = valid_for_dynamic_puppet
			}
		}
