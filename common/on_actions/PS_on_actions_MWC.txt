    on_war_relation_added = {
        effect = {
            # Caso normale - Nazioni indipendenti
            if = {
                limit = {
                    OR = {
                        AND = {
                            ROOT = {
                                OR = {
                                    has_wargoal_against = { target = FROM type = annex_everything }
                                    has_wargoal_against = { target = FROM type = take_state_focus }
                                }
                            }
                            FROM = { capital_scope = { is_claimed_by = ROOT } }
                        }
                        AND = {
                            FROM = {
                                OR = {
                                    has_wargoal_against = { target = ROOT type = annex_everything }
                                    has_wargoal_against = { target = ROOT type = take_state_focus }
                                }
                            }
                            ROOT = { capital_scope = { is_claimed_by = FROM } }
                        }
                    }
                }
                FROM = {
                    every_owned_state = {
                        limit = { is_claimed_by = ROOT }
                        remove_claim_by = ROOT
                    }
                }
                ROOT = {
                    every_owned_state = {
                        limit = { is_claimed_by = FROM }
                        remove_claim_by = FROM
                    }
                }
            }

            # Caso fantocci - Backup per quando FROM è l'Overlord
            if = {
                limit = {
                    FROM = { is_subject = no }
                    ROOT = {
                        OR = {
                            has_wargoal_against = { target = FROM type = annex_everything }
                            has_wargoal_against = { target = FROM type = take_state_focus }
                        }
                    }
                }
                FROM = {
                    every_subject_country = {
                        limit = {
                            capital_scope = { is_claimed_by = ROOT }
                            ROOT = {
                                OR = {
                                    has_wargoal_against = { target = PREV type = annex_everything }
                                    has_wargoal_against = { target = PREV type = take_state_focus }
                                }
                            }
                        }
                        every_owned_state = {
                            limit = { is_claimed_by = ROOT }
                            remove_claim_by = ROOT
                        }
                    }
                }
            }
        }
