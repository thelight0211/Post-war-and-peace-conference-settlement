		}
		if = {
			limit = {
				any_owned_state = {
						NOT = { is_in_home_area = no }
						NOT = { has_state_flag = connected_via_con }

								OR = {
								is_in_home_area = no

							    is_capital = yes
								}
								is_owned_by = ROOT
						}
				}		
			}
			country_event = { id = id.settlement.8 days = 1 }
		}
		if = {
