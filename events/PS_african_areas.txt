country_event = {
    id = ps_african_areas.2
    title = "The fate of Maghreb"
    desc = "How should we administrate the area of Maghreb?"
    is_triggered_only = no
	mean_time_to_happen = { days = 20 }
    trigger = {
        NOT = { has_country_flag = areas2decided }
        is_major = yes
        NOT = { capital_scope = { is_on_continent = africa } }
        has_war = no
        NOT = { 
            OR = {
                original_tag = ENG
                original_tag = FRA
            }
        }
		any_owned_state = {
                count > 3
				OR = {
				is_core_of = ALG
				is_core_of = MOR
				is_core_of = WES
			}
		}
    }

    immediate = {
        set_country_flag = areas2decided 
    }
    
    option = {
        name = "Establish a military administration zone"
        ai_chance = { factor = 95 }
        set_country_flag = Maghreb_decided
    }
    option = {
        name = "Establish separate protectorates"
        ai_chance = { factor = 0 }
        clr_country_flag = MOR_OCCUPIED
        clr_country_flag = RIF_OCCUPIED
        clr_country_flag = ALG_OCCUPIED
        clr_country_flag = WES_OCCUPIED
    }
    option = {
        name = "Keep direct occupation"
        ai_chance = { factor = 0 }
    }
}
