    is_triggered_only = no

            has_war = yes
            is_ai = yes
        }
        overlord = {
            any_enemy_country = {
                NOT = {
                    has_war_with = ROOT
                }
            }
        }

    }
    immediate = {
        overlord = {
            every_enemy_country = {
                limit = {

                            has_war_with = ROOT
                    }

                every_country = {
                    limit = { is_subject_of = PREV.PREV }
                    add_to_war = {
                        targeted_alliance = PREV.PREV
                        enemy = PREV
                        hostility_reason = asked_to_join
                    }
                }
            }
        }
    }
