# 1 - Belgien-Nordfrankreich
country_event = {
	id = rkoccupation.1
	immediate = { log = "[GetDateText]: [Root.GetName]: event rkoccupation.1" }
	title = "Fate of Belgium"
	desc = rkoccupation.1.d
	is_triggered_only = yes
	
	option = {
		name = "Establish Reichskommissariat Belgien-Nordfrankreich"
		ai_chance = { factor = 100 }
		add_political_power = -25
		hidden_effect = {
			if = { limit = { BEL = { is_subject_of = GER } } GER = { annex_country = { target = BEL transfer_troops = yes } } }
		}
		activate_decision = GER_reichskommissariat_belgien_nordfrankreich
	}
	option = {
		name = "Discard the idea"
		trigger = { is_ai = yes }
		ai_chance = { factor = 0 }
		hidden_effect = {
			activate_decision = GER_reichskommissariat_belgien_nordfrankreich
			annex_country = { target = RKB transfer_troops = yes }
			clr_country_flag = BEL_OCCUPIED
		}
	}
	option = {
		name = "We should find another solution"
		trigger = { is_ai = no }
		hidden_effect = {
			clr_country_flag = BEL_OCCUPIED
		}
	}
}
